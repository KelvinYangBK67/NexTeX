\newcommand{\NextUiZhSetName}[2]{%

    {\expandafter\def\csname #1\endcsname{#2}}

}

\newcommand{\NextUiZhApplyDate}{%

          \protect\mbox{\number\year 年\number\month 月\number\day 日}%

        \NextUiZhOriginalDate{\today}%

\newcommand{\NextUiZhApplyChapterFormat}{%
  \renewcommand{\thechapter}{第\zhnumber{\arabic{chapter}}章}%
  \NextUiZhApplySectionNumbering
  \renewcommand{\thefigure}{\arabic{chapter}.\arabic{figure}}%
  \renewcommand{\thetable}{\arabic{chapter}.\arabic{table}}%
  \def\@makechapterhead##1{%
    \vspace*{50\p@}%
    {\parindent \z@ \raggedright \normalfont
      \ifnum \c@secnumdepth >\m@ne
        \if@mainmatter
          \huge\bfseries \thechapter
          \par\nobreak
          \vskip 20\p@
        \fi
      \fi
      \interlinepenalty\@M
      \Huge \bfseries ##1\par\nobreak
      \vskip 40\p@
    }}%
}

\newcommand{\NextUiZhApplySectionNumbering}{%

  \renewcommand{\thesection}{\zhnumber{\arabic{section}}}%
  \renewcommand{\thesubsection}{（\zhnumber{\arabic{subsection}}）}%
  \renewcommand{\thesubsubsection}{\arabic{subsubsection}.}%
}

\newcommand{\NextUiZhApplyTocSpacing}{%

    }%
    \titlecontents{section}
      [2em]
      {}
      {\thecontentslabel\hspace{0.75em}}
      {}
      {\titlerule*[0.8em]{.}\contentspage}%

      [4em]
      {}
      {\thecontentslabel\hspace{0.65em}}
      {}

    \titlecontents{subsubsection}
      [6em]
      {}
      {\thecontentslabel\hspace{0.55em}}
      {}
      {\titlerule*[0.8em]{.}\contentspage}%
  }%
}

\newcommand{\NextUiZhMaybeAddContents}[2]{%
  \ifNextUiZhStarredToc
    \@ifpackageloaded{hyperref}{\phantomsection}{}%
    \addcontentsline{toc}{#1}{#2}%
  \else
    \global\NextUiZhStarredToctrue
  \fi
}
\newcommand{\NextUiZhPatchStarredToc}{%
  \@ifclassloaded{beamer}{}{%
    \let\NextUiZhOriginalTableOfContents\tableofcontents
    \renewcommand{\tableofcontents}{%
      \NoTocNext
      \NextUiZhOriginalTableOfContents
    }%
    \@ifundefined{chapter}{}{%
      \let\NextUiZhOriginalChapter\chapter
      \RenewDocumentCommand{\chapter}{s o m}{%
        \IfBooleanTF{##1}{%
          \NextUiZhOriginalChapter*{##3}%
          \IfNoValueTF{##2}
            {\NextUiZhMaybeAddContents{chapter}{##3}}
            {\NextUiZhMaybeAddContents{chapter}{##2}}%
        }{%
          \IfNoValueTF{##2}
            {\NextUiZhOriginalChapter{##3}}
            {\NextUiZhOriginalChapter[##2]{##3}}%
        }%
      }%
    }%
    \let\NextUiZhOriginalSection\section
    \RenewDocumentCommand{\section}{s o m}{%
      \IfBooleanTF{##1}{%
        \NextUiZhOriginalSection*{##3}%
        \IfNoValueTF{##2}
          {\NextUiZhMaybeAddContents{section}{##3}}
          {\NextUiZhMaybeAddContents{section}{##2}}%
      }{%
        \IfNoValueTF{##2}
          {\NextUiZhOriginalSection{##3}}
          {\NextUiZhOriginalSection[##2]{##3}}%
      }%
    }%
  }%
}
